// ═══════════════════════════════════════════════════
//  IRON COMMAND — Fog of War
// ═══════════════════════════════════════════════════

'use strict';

const Fog = (() => {

  // ─── Create a fresh fog grid ──────────────────────────────────────────────────
  // All tiles start BLACK (unseen)
  function create(cols, rows) {
    return Array.from({ length: rows }, () => new Uint8Array(cols).fill(FOG.BLACK));
  }

  // ─── Reset visible → grey at start of each frame ─────────────────────────────
  // Explored tiles (VISIBLE) become GREY; BLACK stays BLACK
  function resetFrame(fog, cols, rows) {
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        if (fog[r][c] === FOG.VISIBLE) fog[r][c] = FOG.GREY;
      }
    }
  }

  // ─── Reveal tiles around a point ─────────────────────────────────────────────
  // Uses a simple circular reveal with no raycasting (fast, Red Alert style)
  function reveal(fog, col, row, radius, cols, rows) {
    const r0 = Math.max(0, Math.floor(row - radius));
    const r1 = Math.min(rows - 1, Math.ceil(row + radius));
    const c0 = Math.max(0, Math.floor(col - radius));
    const c1 = Math.min(cols - 1, Math.ceil(col + radius));
    const r2 = radius * radius;

    for (let r = r0; r <= r1; r++) {
      const dr = r - row;
      for (let c = c0; c <= c1; c++) {
        const dc = c - col;
        if (dr * dr + dc * dc <= r2) {
          fog[r][c] = FOG.VISIBLE;
        }
      }
    }
  }

  // ─── Full update: recompute all friendly visibility each frame ────────────────
  // units      : array of live player units
  // buildings  : array of all buildings (player + neutral captured)
  // losBonus   : integer added to all LoS radii (from comm towers)
  // grid       : terrain grid (for forest stealth check)
  // cols, rows : map dimensions
  function update(fog, units, buildings, losBonus, grid, cols, rows) {
    resetFrame(fog, cols, rows);

    const bonus = losBonus || 0;

    // Player units reveal around themselves
    for (const u of units) {
      if (u.dead) continue;
      const def = getUnitDef(u.type);
      if (!def) continue;
      const col = Math.floor(u.px / CELL);
      const row = Math.floor(u.py / CELL);
      reveal(fog, col, row, (def.los || 3) + bonus, cols, rows);
    }

    // Player buildings reveal around themselves
    for (const b of buildings) {
      if (b.dead) continue;
      // Skip enemy buildings
      if (isEnemyType(b.type)) continue;
      // Neutral buildings only reveal if captured by player
      if (isNeutralType(b.type) && b.captureTeam !== 'player') continue;

      const def = getBldgDef(b.type);
      if (!def) continue;
      reveal(fog, b.col, b.row, (def.los || 2) + bonus, cols, rows);
    }
  }

  // ─── Query helpers ────────────────────────────────────────────────────────────

  function isVisible(fog, col, row) {
    if (!fog[row]) return false;
    return fog[row][col] === FOG.VISIBLE;
  }

  function isExplored(fog, col, row) {
    if (!fog[row]) return false;
    return fog[row][col] >= FOG.GREY;
  }

  function isBlack(fog, col, row) {
    if (!fog[row]) return true;
    return fog[row][col] === FOG.BLACK;
  }

  // Check whether a world-pixel position is currently visible
  function isPxVisible(fog, wx, wy) {
    const col = Math.floor(wx / CELL);
    const row = Math.floor(wy / CELL);
    return isVisible(fog, col, row);
  }

  // ─── Shroud rendering (Red Alert style) ──────────────────────────────────────
  // Draws the fog overlay onto the canvas.
  // Call this AFTER all terrain, buildings, and units have been drawn.
  //
  // scrollX, scrollY : current viewport scroll in world pixels
  // viewW, viewH     : viewport size in pixels
  // cols, rows       : map dimensions

  function render(ctx, fog, scrollX, scrollY, viewW, viewH, cols, rows) {
    const startC = Math.max(0, Math.floor(scrollX / CELL));
    const endC   = Math.min(cols - 1, Math.ceil((scrollX + viewW) / CELL));
    const startR = Math.max(0, Math.floor(scrollY / CELL));
    const endR   = Math.min(rows - 1, Math.ceil((scrollY + viewH) / CELL));

    for (let r = startR; r <= endR; r++) {
      for (let c = startC; c <= endC; c++) {
        const state = fog[r]?.[c] ?? FOG.BLACK;
        if (state === FOG.VISIBLE) continue; // fully clear — skip

        const sx = c * CELL - scrollX;
        const sy = r * CELL - scrollY;

        if (state === FOG.BLACK) {
          // Fully unexplored — solid black with RA-style pixel border hint
          ctx.fillStyle = '#000000';
          ctx.fillRect(sx, sy, CELL, CELL);

          // Check neighbours: if an adjacent tile is explored, draw
          // a subtle "shroud edge" gradient strip (pixel-art style)
          const hasExploredNeighbour =
            isExplored(fog, c + 1, r) ||
            isExplored(fog, c - 1, r) ||
            isExplored(fog, c, r + 1) ||
            isExplored(fog, c, r - 1);

          if (hasExploredNeighbour) {
            // Soft edge: checkerboard-like dither strip
            ctx.fillStyle = 'rgba(20,20,40,0.85)';
            // Right edge dither
            if (isExplored(fog, c + 1, r)) {
              for (let py = 0; py < CELL; py += 2) {
                ctx.fillRect(sx + CELL - 4, sy + py, 4, 1);
              }
            }
            // Bottom edge dither
            if (isExplored(fog, c, r + 1)) {
              for (let px2 = 0; px2 < CELL; px2 += 2) {
                ctx.fillRect(sx + px2, sy + CELL - 4, 1, 4);
              }
            }
            // Left edge dither
            if (isExplored(fog, c - 1, r)) {
              for (let py = 0; py < CELL; py += 2) {
                ctx.fillRect(sx, sy + py, 4, 1);
              }
            }
            // Top edge dither
            if (isExplored(fog, c, r - 1)) {
              for (let px2 = 0; px2 < CELL; px2 += 2) {
                ctx.fillRect(sx + px2, sy, 1, 4);
              }
            }
          }

        } else {
          // GREY — previously explored but not currently visible
          // Dark blue-grey tint overlay (RA uses a dark semi-transparent wash)
          ctx.fillStyle = 'rgba(8,10,28,0.62)';
          ctx.fillRect(sx, sy, CELL, CELL);

          // Pixel-dither pattern to give it that classic RA shroud texture
          ctx.fillStyle = 'rgba(0,0,20,0.28)';
          for (let py = 0; py < CELL; py += 2) {
            for (let px2 = py % 4 === 0 ? 0 : 2; px2 < CELL; px2 += 4) {
              ctx.fillRect(sx + px2, sy + py, 2, 1);
            }
          }
        }
      }
    }
  }

  // ─── Minimap fog colours ──────────────────────────────────────────────────────
  // Returns the colour to draw on the minimap for a fog state + terrain
  function minimapColor(fogState, terrainColor) {
    if (fogState === FOG.BLACK)   return '#000000';
    if (fogState === FOG.GREY)    return _darkenHex(terrainColor, 0.38);
    return terrainColor; // VISIBLE — full colour
  }

  // Darken a hex colour by a factor (0 = black, 1 = original)
  function _darkenHex(hex, factor) {
    const r = parseInt(hex.slice(1, 3), 16);
    const g = parseInt(hex.slice(3, 5), 16);
    const b = parseInt(hex.slice(5, 7), 16);
    const dr = Math.round(r * factor);
    const dg = Math.round(g * factor);
    const db = Math.round(b * factor);
    return `rgb(${dr},${dg},${db})`;
  }

  // ─── Serialise / deserialise ──────────────────────────────────────────────────
  // Convert fog grid to a compact array for save-state snapshots
  function serialise(fog) {
    return fog.map(row => Array.from(row));
  }

  function deserialise(data, cols, rows) {
    const fog = create(cols, rows);
    for (let r = 0; r < Math.min(rows, data.length); r++) {
      for (let c = 0; c < Math.min(cols, data[r].length); c++) {
        fog[r][c] = data[r][c];
      }
    }
    return fog;
  }

  return {
    create,
    resetFrame,
    reveal,
    update,
    isVisible,
    isExplored,
    isBlack,
    isPxVisible,
    render,
    minimapColor,
    serialise,
    deserialise,
  };

})();
